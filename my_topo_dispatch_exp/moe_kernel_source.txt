static torch::Tensor hw_pack_pool;
static torch::Tensor hw_expand_pool;
static torch::Tensor hw_cnt_pool;

torch::Tensor moe_hw_pack_npu(const torch::Tensor &x, const torch::Tensor &expert_ids, int64_t ep_world_size, int64_t moe_expert_num, int64_t chunk_rows)
{
    const c10::OptionalDeviceGuard guard(x.device());
    auto x_c = x.contiguous(); auto idx_c = expert_ids.contiguous();
    int64_t req_rows = ep_world_size * chunk_rows, h = x_c.size(1);
    if (!hw_pack_pool.defined() || hw_pack_pool.device() != x.device() || hw_pack_pool.size(0) < req_rows || hw_pack_pool.size(1) != h) {
        hw_pack_pool = torch::empty({req_rows, h}, x.options());
    }
    auto out = hw_pack_pool.narrow(0, 0, req_rows);
    int64_t bs = idx_c.size(0), k = idx_c.size(1);
    int64_t numBlocks = std::min(get_hw_aiv_num(), bs * k);
    auto stream = c10_npu::getCurrentNPUStream().stream(false);
    auto xp = (GM_ADDR)x_c.data_ptr(), ip = (GM_ADDR)idx_c.data_ptr(), op = (GM_ADDR)out.data_ptr();
    at_npu::native::OpCommand::RunOpApi("MoeHwPack", [=]() -> int { launch_moe_hw_pack_bf16(xp, ip, op, bs, k, h, ep_world_size, moe_expert_num, chunk_rows, numBlocks, stream); return 0; });
    return out;
}

std::tuple<torch::Tensor, torch::Tensor> moe_hw_unpack_npu(const torch::Tensor &recv_buf, int64_t ep_world_size, int64_t moe_expert_num, int64_t chunk_rows, int64_t max_capacity)
{
    const c10::OptionalDeviceGuard guard(recv_buf.device());
    auto r_c = recv_buf.contiguous();
    int64_t h = r_c.size(1), local_exp = moe_expert_num / ep_world_size;
    if (!hw_expand_pool.defined() || hw_expand_pool.device() != recv_buf.device() || hw_expand_pool.size(0) < max_capacity || hw_expand_pool.size(1) != h) {
        hw_expand_pool = torch::empty({max_capacity, h}, recv_buf.options());
    }
    if (!hw_cnt_pool.defined() || hw_cnt_pool.device() != recv_buf.device() || hw_cnt_pool.size(0) < local_exp) {
        hw_cnt_pool = torch::empty({local_exp}, recv_buf.options().dtype(torch::kInt32));
    }
    auto expand_x = hw_expand_pool.narrow(0, 0, max_capacity);
    auto ep_cnt = hw_cnt_pool.narrow(0, 0, local_exp);
    int64_t numBlocks = get_hw_aiv_num();
    auto stream = c10_npu::getCurrentNPUStream().stream(false);
    auto ip = (GM_ADDR)r_c.data_ptr(), op = (GM_ADDR)expand_x.data_ptr(), cp = (GM_ADDR)ep_cnt.data_ptr();
    at_npu::native::OpCommand::RunOpApi("MoeHwUnpack", [=]() -> int { launch_moe_hw_unpack_bf16(ip, op, cp, h, ep_world_size, moe_expert_num, chunk_rows, numBlocks, stream); return 0; });
    return std::make_tuple(expand_x, ep_cnt);
}

constexpr static int32_t BUFFER_NUM = 2;
constexpr static int64_t MAX_TILE_BYTES = 32768;
constexpr static int64_t MAX_PAIRS_UB = 8192;
constexpr static int32_t MAX_EXPERTS = 128;
constexpr static int32_t MAX_RANKS = 16;

__global__ __aicore__ void moe_hw_pack_bf16_kernel(
    GM_ADDR x, GM_ADDR expert_ids, GM_ADDR packed_out,
    int64_t bs, int64_t k, int64_t hidden_size,
    int64_t ep_world_size, int64_t moe_expert_num, int64_t chunk_rows)
{
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> idxBuf;

    int64_t total_pairs = bs * k;
    int64_t block_idx = AscendC::GetBlockIdx();
    int64_t block_num = AscendC::GetBlockNum();
    int64_t pairs_per_core = (total_pairs + block_num - 1) / block_num;
    int64_t start_pair = block_idx * pairs_per_core;
    int64_t end_pair = (start_pair + pairs_per_core) < total_pairs ? (start_pair + pairs_per_core) : total_pairs;
    if (start_pair >= end_pair) return;

    int64_t max_elems = MAX_TILE_BYTES / (int64_t)sizeof(bfloat16_t);
    int64_t tile_elems = hidden_size < max_elems ? hidden_size : max_elems;
    uint32_t alloc_bytes = static_cast<uint32_t>(tile_elems * sizeof(bfloat16_t));
    pipe.InitBuffer(inQueue, BUFFER_NUM, alloc_bytes);
    pipe.InitBuffer(outQueue, BUFFER_NUM, alloc_bytes);
    pipe.InitBuffer(idxBuf, MAX_PAIRS_UB * sizeof(int32_t));

    AscendC::GlobalTensor<bfloat16_t> xGm, outGm;
    AscendC::GlobalTensor<int32_t> expIdGm;
    xGm.SetGlobalBuffer((__gm__ bfloat16_t*)x);
    expIdGm.SetGlobalBuffer((__gm__ int32_t*)expert_ids);
    outGm.SetGlobalBuffer((__gm__ bfloat16_t*)packed_out);

    // 硬件优化 1：整块向量化读入 expert_ids 至高速片上 UB，彻底消除 20 核抢占 GM 标量总线
    auto localExp = idxBuf.Get<int32_t>();
    int64_t aligned_pairs = ((total_pairs + 7) / 8) * 8;
    if (aligned_pairs > MAX_PAIRS_UB) aligned_pairs = MAX_PAIRS_UB;
    AscendC::DataCopyExtParams idxCopy{1, static_cast<uint32_t>(total_pairs * sizeof(int32_t)), 0, 0, 0};
    AscendC::DataCopyPadExtParams<int32_t> idxPad{true, 0, static_cast<uint8_t>(aligned_pairs - total_pairs), 0};
    AscendC::DataCopyPad(localExp, expIdGm, idxCopy, idxPad);
    AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
    AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);

    int32_t experts_per_rank = static_cast<int32_t>(moe_expert_num / ep_world_size);
    int32_t exp_counts[MAX_EXPERTS];
    int32_t prefix_before[MAX_EXPERTS];
    for (int32_t e = 0; e < moe_expert_num; ++e) { exp_counts[e] = 0; prefix_before[e] = 0; }

    for (int64_t p = 0; p < total_pairs; ++p) {
        int32_t eid = localExp.GetValue(p);
        if (p < start_pair) prefix_before[eid]++;
        exp_counts[eid]++;
    }

    int32_t exp_dst_base[MAX_EXPERTS];
    for (int32_t r = 0; r < ep_world_size; ++r) {
        int32_t acc = 0;
        for (int32_t le = 0; le < experts_per_rank; ++le) {
            int32_t gid = r * experts_per_rank + le;
            exp_dst_base[gid] = static_cast<int32_t>(r * chunk_rows) + acc;
            acc += exp_counts[gid];
        }
        if (block_idx == 0) {
            __gm__ int32_t* meta_ptr = reinterpret_cast<__gm__ int32_t*>(
                reinterpret_cast<__gm__ bfloat16_t*>(packed_out) + (r * chunk_rows + (chunk_rows - 1)) * hidden_size);
            for (int32_t le = 0; le < experts_per_rank; ++le) meta_ptr[le] = exp_counts[r * experts_per_rank + le];
        }
    }

    // 硬件优化 2：严格遵循 MTE2 -> Vector -> MTE3 双队列流水线状态机，100% 零竞态误差
    AscendC::DataCopyPadExtParams<bfloat16_t> padParams{false, 0, 0, 0};
    for (int64_t p = start_pair; p < end_pair; ++p) {
        int32_t eid = localExp.GetValue(p);
        int32_t dst_slot = exp_dst_base[eid] + (prefix_before[eid]++);
        int64_t src_base = (p / k) * hidden_size;
        int64_t dst_base = static_cast<int64_t>(dst_slot) * hidden_size;

        for (int64_t offset = 0; offset < hidden_size; offset += tile_elems) {
            int64_t cur_elems = (hidden_size - offset) < tile_elems ? (hidden_size - offset) : tile_elems;
            AscendC::DataCopyExtParams copyParams{1, static_cast<uint32_t>(cur_elems * sizeof(bfloat16_t)), 0, 0, 0};
            auto localIn = inQueue.AllocTensor<bfloat16_t>();
            AscendC::DataCopyPad(localIn, xGm[src_base + offset], copyParams, padParams);
            inQueue.EnQue(localIn);
            localIn = inQueue.DeQue<bfloat16_t>();
            auto localOut = outQueue.AllocTensor<bfloat16_t>();
            AscendC::DataCopy(localOut, localIn, cur_elems);
            outQueue.EnQue(localOut);
            inQueue.FreeTensor(localIn);
            localOut = outQueue.DeQue<bfloat16_t>();
            AscendC::DataCopyPad(outGm[dst_base + offset], localOut, copyParams);
            outQueue.FreeTensor(localOut);
        }
    }
}

__global__ __aicore__ void moe_hw_unpack_bf16_kernel(
    GM_ADDR recv_buf, GM_ADDR expand_out, GM_ADDR ep_recv_counts,
    int64_t hidden_size, int64_t ep_world_size, int64_t moe_expert_num, int64_t chunk_rows)
{
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueue;

    int64_t block_idx = AscendC::GetBlockIdx();
    int64_t block_num = AscendC::GetBlockNum();
    int64_t max_elems = MAX_TILE_BYTES / (int64_t)sizeof(bfloat16_t);
    int64_t tile_elems = hidden_size < max_elems ? hidden_size : max_elems;
    uint32_t alloc_bytes = static_cast<uint32_t>(tile_elems * sizeof(bfloat16_t));
    pipe.InitBuffer(inQueue, BUFFER_NUM, alloc_bytes);
    pipe.InitBuffer(outQueue, BUFFER_NUM, alloc_bytes);

    AscendC::GlobalTensor<bfloat16_t> inGm, outGm;
    inGm.SetGlobalBuffer((__gm__ bfloat16_t*)recv_buf);
    outGm.SetGlobalBuffer((__gm__ bfloat16_t*)expand_out);

    int32_t experts_per_rank = static_cast<int32_t>(moe_expert_num / ep_world_size);
    int32_t recv_cnt[MAX_RANKS][MAX_EXPERTS];
    int32_t src_off[MAX_RANKS][MAX_EXPERTS];

    for (int32_t r = 0; r < ep_world_size; ++r) {
        __gm__ int32_t* meta_ptr = reinterpret_cast<__gm__ int32_t*>(
            reinterpret_cast<__gm__ bfloat16_t*>(recv_buf) + (r * chunk_rows + (chunk_rows - 1)) * hidden_size);
        int32_t acc = static_cast<int32_t>(r * chunk_rows);
        for (int32_t le = 0; le < experts_per_rank; ++le) {
            int32_t c = meta_ptr[le];
            recv_cnt[r][le] = c;
            src_off[r][le] = acc;
            acc += c;
        }
    }

    int32_t dst_off[MAX_EXPERTS][MAX_RANKS];
    int32_t total_recv_tokens = 0;
    for (int32_t le = 0; le < experts_per_rank; ++le) {
        int32_t exp_sum = 0;
        for (int32_t r = 0; r < ep_world_size; ++r) {
            dst_off[le][r] = total_recv_tokens;
            total_recv_tokens += recv_cnt[r][le];
            exp_sum += recv_cnt[r][le];
        }
        if (block_idx == 0) reinterpret_cast<__gm__ int32_t*>(ep_recv_counts)[le] = exp_sum;
    }
    if (total_recv_tokens == 0) return;

    int64_t tokens_per_core = (total_recv_tokens + block_num - 1) / block_num;
    int64_t start_tok = block_idx * tokens_per_core;
    int64_t end_tok = (start_tok + tokens_per_core) < total_recv_tokens ? (start_tok + tokens_per_core) : total_recv_tokens;
    if (start_tok >= end_tok) return;

    // 硬件优化 3：按目标地址连续递增顺序写出（Write-Combining 写合并满带宽）
    AscendC::DataCopyPadExtParams<bfloat16_t> padParams{false, 0, 0, 0};
    for (int32_t le = 0; le < experts_per_rank; ++le) {
        for (int32_t r = 0; r < ep_world_size; ++r) {
            int32_t cnt = recv_cnt[r][le];
            if (cnt <= 0) continue;
            int64_t blk_dst_start = dst_off[le][r];
            int64_t ov_start = blk_dst_start > start_tok ? blk_dst_start : start_tok;
            int64_t ov_end = (blk_dst_start + cnt) < end_tok ? (blk_dst_start + cnt) : end_tok;
            for (int64_t t = ov_start; t < ov_end; ++t) {
                int64_t src_base = (static_cast<int64_t>(src_off[r][le]) + (t - blk_dst_start)) * hidden_size;
                int64_t dst_base = t * hidden_size;
                for (int64_t offset = 0; offset < hidden_size; offset += tile_elems) {
                    int64_t cur_elems = (hidden_size - offset) < tile_elems ? (hidden_size - offset) : tile_elems;
                    AscendC::DataCopyExtParams copyParams{1, static_cast<uint32_t>(cur_elems * sizeof(bfloat16_t)), 0, 0, 0};
                    auto localIn = inQueue.AllocTensor<bfloat16_t>();
                    AscendC::DataCopyPad(localIn, inGm[src_base + offset], copyParams, padParams);
                    inQueue.EnQue(localIn);
                    localIn = inQueue.DeQue<bfloat16_t>();
                    auto localOut = outQueue.AllocTensor<bfloat16_t>();
                    AscendC::DataCopy(localOut, localIn, cur_elems);
                    outQueue.EnQue(localOut);
                    inQueue.FreeTensor(localIn);
                    localOut = outQueue.DeQue<bfloat16_t>();
                    AscendC::DataCopyPad(outGm[dst_base + offset], localOut, copyParams);
                    outQueue.FreeTensor(localOut);
                }
            }
        }
    }
}

int64_t get_hw_aiv_num()
{
    auto platform = platform_ascendc::PlatformAscendCManager::GetInstance();
    int64_t coreNum = platform->GetCoreNumAiv();
    return coreNum > 0 ? coreNum : 20;
}

// 静态持久化缓冲池（消除每次调用的大显存分配与清零延迟）
static torch::Tensor g_pack_pool;
static torch::Tensor g_expand_pool;
static torch::Tensor g_cnt_pool;

torch::Tensor moe_e2e_pack_npu(const torch::Tensor &x, const torch::Tensor &expert_ids, int64_t ep_world_size, int64_t moe_expert_num, int64_t chunk_rows)
{
    const c10::OptionalDeviceGuard guard(x.device());
    auto x_c = x.contiguous(); auto idx_c = expert_ids.contiguous();
    int64_t req_rows = ep_world_size * chunk_rows, h = x_c.size(1);
    if (!g_pack_pool.defined() || g_pack_pool.device() != x.device() || g_pack_pool.size(0) < req_rows || g_pack_pool.size(1) != h || g_pack_pool.scalar_type() != x.scalar_type()) {
        g_pack_pool = torch::empty({req_rows, h}, x.options());
    }
    auto out = g_pack_pool.narrow(0, 0, req_rows);
    int64_t bs = idx_c.size(0), k = idx_c.size(1);
    int64_t numBlocks = std::min(get_e2e_aiv_num(), bs * k);
    auto stream = c10_npu::getCurrentNPUStream().stream(false);
    auto xp = (GM_ADDR)x_c.data_ptr(), ip = (GM_ADDR)idx_c.data_ptr(), op = (GM_ADDR)out.data_ptr();
    at_npu::native::OpCommand::RunOpApi("MoeE2ePack", [=]() -> int { launch_moe_e2e_pack_bf16(xp, ip, op, bs, k, h, ep_world_size, moe_expert_num, chunk_rows, numBlocks, stream); return 0; });
    return out;
}

    const c10::OptionalDeviceGuard guard(recv_buf.device());
    auto r_c = recv_buf.contiguous();
    int64_t h = r_c.size(1), local_exp = moe_expert_num / ep_world_size;
    if (!g_expand_pool.defined() || g_expand_pool.device() != recv_buf.device() || g_expand_pool.size(0) < max_capacity || g_expand_pool.size(1) != h || g_expand_pool.scalar_type() != recv_buf.scalar_type()) {
        g_expand_pool = torch::empty({max_capacity, h}, recv_buf.options());
    }
    if (!g_cnt_pool.defined() || g_cnt_pool.device() != recv_buf.device() || g_cnt_pool.size(0) < local_exp) {
        g_cnt_pool = torch::empty({local_exp}, recv_buf.options().dtype(torch::kInt32));
    }
    auto expand_x = g_expand_pool.narrow(0, 0, max_capacity);
    auto ep_cnt = g_cnt_pool.narrow(0, 0, local_exp);
    int64_t numBlocks = get_e2e_aiv_num();
    auto stream = c10_npu::getCurrentNPUStream().stream(false);
    auto ip = (GM_ADDR)r_c.data_ptr(), op = (GM_ADDR)expand_x.data_ptr(), cp = (GM_ADDR)ep_cnt.data_ptr();
    at_npu::native::OpCommand::RunOpApi("MoeE2eUnpack", [=]() -> int { launch_moe_e2e_unpack_bf16(ip, op, cp, h, ep_world_size, moe_expert_num, chunk_rows, numBlocks, stream); return 0; });
    return std::make_tuple(expand_x, ep_cnt);
}

#include "kernel_operator.h"
#include "platform/platform_ascendc.h"

constexpr static int32_t BUFFER_NUM = 2;
constexpr static int64_t MAX_TILE_BYTES = 32768;
constexpr static int32_t MAX_EXPERTS = 128;
constexpr static int32_t MAX_RANKS = 16;

__global__ __aicore__ void moe_e2e_pack_bf16_kernel(
    GM_ADDR x, GM_ADDR expert_ids, GM_ADDR packed_out,
    int64_t bs, int64_t k, int64_t hidden_size,
    int64_t ep_world_size, int64_t moe_expert_num, int64_t chunk_rows)
{
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueue;

    int64_t total_pairs = bs * k;
    int64_t block_idx = AscendC::GetBlockIdx();
    int64_t block_num = AscendC::GetBlockNum();
    int64_t pairs_per_core = (total_pairs + block_num - 1) / block_num;
    int64_t start_pair = block_idx * pairs_per_core;
    int64_t end_pair = (start_pair + pairs_per_core) < total_pairs ? (start_pair + pairs_per_core) : total_pairs;
    if (start_pair >= end_pair) return;

    int64_t max_elems = MAX_TILE_BYTES / (int64_t)sizeof(bfloat16_t);
    int64_t tile_elems = hidden_size < max_elems ? hidden_size : max_elems;
    uint32_t alloc_bytes = static_cast<uint32_t>(tile_elems * sizeof(bfloat16_t));
    pipe.InitBuffer(inQueue, BUFFER_NUM, alloc_bytes);
    pipe.InitBuffer(outQueue, BUFFER_NUM, alloc_bytes);

    AscendC::GlobalTensor<bfloat16_t> xGm, outGm;
    AscendC::GlobalTensor<int32_t> expIdGm;
    xGm.SetGlobalBuffer((__gm__ bfloat16_t*)x);
    expIdGm.SetGlobalBuffer((__gm__ int32_t*)expert_ids);
    outGm.SetGlobalBuffer((__gm__ bfloat16_t*)packed_out);

    int32_t experts_per_rank = static_cast<int32_t>(moe_expert_num / ep_world_size);
    int32_t exp_counts[MAX_EXPERTS];
    int32_t prefix_before[MAX_EXPERTS];
    for (int32_t e = 0; e < moe_expert_num; ++e) { exp_counts[e] = 0; prefix_before[e] = 0; }

    for (int64_t p = 0; p < total_pairs; ++p) {
        int32_t eid = expIdGm.GetValue(p);
        if (p < start_pair) prefix_before[eid]++;
        exp_counts[eid]++;
    }

    int32_t exp_dst_base[MAX_EXPERTS];
    for (int32_t r = 0; r < ep_world_size; ++r) {
        int32_t acc = 0;
        for (int32_t le = 0; le < experts_per_rank; ++le) {
            int32_t gid = r * experts_per_rank + le;
            exp_dst_base[gid] = static_cast<int32_t>(r * chunk_rows) + acc;
            acc += exp_counts[gid];
        }
        if (block_idx == 0) {
            __gm__ int32_t* meta_ptr = reinterpret_cast<__gm__ int32_t*>(
                reinterpret_cast<__gm__ bfloat16_t*>(packed_out) + (r * chunk_rows + (chunk_rows - 1)) * hidden_size);
            for (int32_t le = 0; le < experts_per_rank; ++le) meta_ptr[le] = exp_counts[r * experts_per_rank + le];
        }
    }

    AscendC::DataCopyPadExtParams<bfloat16_t> padParams{false, 0, 0, 0};
    for (int64_t p = start_pair; p < end_pair; ++p) {
        int32_t eid = expIdGm.GetValue(p);
        int32_t dst_slot = exp_dst_base[eid] + (prefix_before[eid]++);
        int64_t src_base = (p / k) * hidden_size;
        int64_t dst_base = static_cast<int64_t>(dst_slot) * hidden_size;

        for (int64_t offset = 0; offset < hidden_size; offset += tile_elems) {
            int64_t cur_elems = (hidden_size - offset) < tile_elems ? (hidden_size - offset) : tile_elems;
            AscendC::DataCopyExtParams copyParams{1, static_cast<uint32_t>(cur_elems * sizeof(bfloat16_t)), 0, 0, 0};
            auto localIn = inQueue.AllocTensor<bfloat16_t>();
            AscendC::DataCopyPad(localIn, xGm[src_base + offset], copyParams, padParams);
            inQueue.EnQue(localIn);
            localIn = inQueue.DeQue<bfloat16_t>();
            auto localOut = outQueue.AllocTensor<bfloat16_t>();
            AscendC::DataCopy(localOut, localIn, cur_elems);
            outQueue.EnQue(localOut);
            inQueue.FreeTensor(localIn);
            localOut = outQueue.DeQue<bfloat16_t>();
            AscendC::DataCopyPad(outGm[dst_base + offset], localOut, copyParams);
            outQueue.FreeTensor(localOut);
        }
    }
}

__global__ __aicore__ void moe_e2e_unpack_bf16_kernel(
    GM_ADDR recv_buf, GM_ADDR expand_out, GM_ADDR ep_recv_counts,
    int64_t hidden_size, int64_t ep_world_size, int64_t moe_expert_num, int64_t chunk_rows)
{
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueue;

    int64_t block_idx = AscendC::GetBlockIdx();
    int64_t block_num = AscendC::GetBlockNum();
    int64_t max_elems = MAX_TILE_BYTES / (int64_t)sizeof(bfloat16_t);
    int64_t tile_elems = hidden_size < max_elems ? hidden_size : max_elems;
    uint32_t alloc_bytes = static_cast<uint32_t>(tile_elems * sizeof(bfloat16_t));
    pipe.InitBuffer(inQueue, BUFFER_NUM, alloc_bytes);
    pipe.InitBuffer(outQueue, BUFFER_NUM, alloc_bytes);

    AscendC::GlobalTensor<bfloat16_t> inGm, outGm;
    inGm.SetGlobalBuffer((__gm__ bfloat16_t*)recv_buf);
    outGm.SetGlobalBuffer((__gm__ bfloat16_t*)expand_out);

    int32_t experts_per_rank = static_cast<int32_t>(moe_expert_num / ep_world_size);
    int32_t recv_cnt[MAX_RANKS][MAX_EXPERTS];
    int32_t src_off[MAX_RANKS][MAX_EXPERTS];

    for (int32_t r = 0; r < ep_world_size; ++r) {
        __gm__ int32_t* meta_ptr = reinterpret_cast<__gm__ int32_t*>(
            reinterpret_cast<__gm__ bfloat16_t*>(recv_buf) + (r * chunk_rows + (chunk_rows - 1)) * hidden_size);
        int32_t acc = static_cast<int32_t>(r * chunk_rows);
        for (int32_t le = 0; le < experts_per_rank; ++le) {
            int32_t c = meta_ptr[le];
            recv_cnt[r][le] = c;
            src_off[r][le] = acc;
            acc += c;
        }
    }

    int32_t dst_off[MAX_EXPERTS][MAX_RANKS];
    int32_t total_recv_tokens = 0;
    for (int32_t le = 0; le < experts_per_rank; ++le) {
        int32_t exp_sum = 0;
        for (int32_t r = 0; r < ep_world_size; ++r) {
            dst_off[le][r] = total_recv_tokens;
            total_recv_tokens += recv_cnt[r][le];
            exp_sum += recv_cnt[r][le];
        }
        if (block_idx == 0) reinterpret_cast<__gm__ int32_t*>(ep_recv_counts)[le] = exp_sum;
    }
    if (total_recv_tokens == 0) return;

    int64_t tokens_per_core = (total_recv_tokens + block_num - 1) / block_num;
    int64_t start_tok = block_idx * tokens_per_core;
    int64_t end_tok = (start_tok + tokens_per_core) < total_recv_tokens ? (start_tok + tokens_per_core) : total_recv_tokens;
    if (start_tok >= end_tok) return;

    AscendC::DataCopyPadExtParams<bfloat16_t> padParams{false, 0, 0, 0};
    for (int32_t le = 0; le < experts_per_rank; ++le) {
        for (int32_t r = 0; r < ep_world_size; ++r) {
            int32_t cnt = recv_cnt[r][le];
            if (cnt <= 0) continue;
            int64_t blk_dst_start = dst_off[le][r];
            int64_t ov_start = blk_dst_start > start_tok ? blk_dst_start : start_tok;
            int64_t ov_end = (blk_dst_start + cnt) < end_tok ? (blk_dst_start + cnt) : end_tok;
            for (int64_t t = ov_start; t < ov_end; ++t) {
                int64_t src_base = (static_cast<int64_t>(src_off[r][le]) + (t - blk_dst_start)) * hidden_size;
                int64_t dst_base = t * hidden_size;
                for (int64_t offset = 0; offset < hidden_size; offset += tile_elems) {
                    int64_t cur_elems = (hidden_size - offset) < tile_elems ? (hidden_size - offset) : tile_elems;
                    AscendC::DataCopyExtParams copyParams{1, static_cast<uint32_t>(cur_elems * sizeof(bfloat16_t)), 0, 0, 0};
                    auto localIn = inQueue.AllocTensor<bfloat16_t>();
                    AscendC::DataCopyPad(localIn, inGm[src_base + offset], copyParams, padParams);
                    inQueue.EnQue(localIn);
                    localIn = inQueue.DeQue<bfloat16_t>();
                    auto localOut = outQueue.AllocTensor<bfloat16_t>();
                    AscendC::DataCopy(localOut, localIn, cur_elems);
                    outQueue.EnQue(localOut);
                    inQueue.FreeTensor(localIn);
                    localOut = outQueue.DeQue<bfloat16_t>();
                    AscendC::DataCopyPad(outGm[dst_base + offset], localOut, copyParams);
                    outQueue.FreeTensor(localOut);
                }
            }
        }
    }
}

int64_t get_e2e_aiv_num()
{
    auto platform = platform_ascendc::PlatformAscendCManager::GetInstance();
    int64_t coreNum = platform->GetCoreNumAiv();
    return coreNum > 0 ? (coreNum < 20 ? coreNum : 20) : 20;
}

torch::Tensor add_npu(const torch::Tensor &x, const torch::Tensor &y)
{
    const c10::OptionalDeviceGuard guard(x.device());
    auto z = add_meta(x, y);
    auto stream = c10_npu::getCurrentNPUStream().stream(false);
    int64_t totalLength = x.numel();
    int64_t numBlocks, blockLength, tileSize;
    std::tie(numBlocks, blockLength, tileSize) = calc_add_tiling_params(totalLength);
    auto x_ptr = (GM_ADDR)x.data_ptr();
    auto y_ptr = (GM_ADDR)y.data_ptr();
    auto z_ptr = (GM_ADDR)z.data_ptr();

    auto acl_call = [=]() -> int {
        if (x.scalar_type() == torch::kFloat32) {
            ADD_KERNEL_LAUNCH_FLOAT(x_ptr, y_ptr, z_ptr, totalLength, numBlocks, blockLength, tileSize, stream);
        } else if (x.scalar_type() == torch::kFloat16) {
            ADD_KERNEL_LAUNCH_HALF(x_ptr, y_ptr, z_ptr, totalLength, numBlocks, blockLength, tileSize, stream);
        } else if (x.scalar_type() == torch::kInt32) {
            ADD_KERNEL_LAUNCH_INT32(x_ptr, y_ptr, z_ptr, totalLength, numBlocks, blockLength, tileSize, stream);
        }
        return 0;
    };
    at_npu::native::OpCommand::RunOpApi("Add", acl_call);
    return z;
}

#include <tuple>

constexpr static int64_t PIPELINE_DEPTH = 2;

// Kernel implementation
template <typename T>
__global__ __aicore__ void add_kernel(GM_ADDR x, GM_ADDR y, GM_ADDR z, int64_t totalLength, int64_t blockLength, uint32_t tileSize)
{
    AscendC::TPipe pipe;
    AscendC::GlobalTensor<T> xGm, yGm, zGm;
    AscendC::TQue<AscendC::QuePosition::VECIN, PIPELINE_DEPTH> inQueueX;
    AscendC::TQue<AscendC::QuePosition::VECIN, PIPELINE_DEPTH> inQueueY;
    AscendC::TQue<AscendC::QuePosition::VECOUT, PIPELINE_DEPTH> outQueueZ;
    pipe.InitBuffer(inQueueX, PIPELINE_DEPTH, tileSize);
    pipe.InitBuffer(inQueueY, PIPELINE_DEPTH, tileSize);
    pipe.InitBuffer(outQueueZ, PIPELINE_DEPTH, tileSize);
    xGm.SetGlobalBuffer((__gm__ T *)x + blockLength * AscendC::GetBlockIdx());
    yGm.SetGlobalBuffer((__gm__ T *)y + blockLength * AscendC::GetBlockIdx());
    zGm.SetGlobalBuffer((__gm__ T *)z + blockLength * AscendC::GetBlockIdx());

    int64_t currentBlockLength = totalLength - AscendC::GetBlockIdx() * blockLength;
    if (currentBlockLength > blockLength) currentBlockLength = blockLength;
    int64_t elementNumPerTile = tileSize / sizeof(T);
    int64_t tileNum = currentBlockLength / elementNumPerTile;
    int64_t tailTileElementNum = currentBlockLength - tileNum * elementNumPerTile;

    for (int64_t i = 0; i < tileNum; ++i) {
        int64_t offset = i * elementNumPerTile;
        AscendC::DataCopyExtParams copyParams{1, static_cast<uint32_t>(elementNumPerTile * sizeof(T)), 0, 0, 0};
        AscendC::DataCopyPadExtParams<T> padParams{false, 0, 0, 0};
        auto xLocal = inQueueX.AllocTensor<T>();
        auto yLocal = inQueueY.AllocTensor<T>();
        AscendC::DataCopyPad(xLocal, xGm[offset], copyParams, padParams);
        AscendC::DataCopyPad(yLocal, yGm[offset], copyParams, padParams);
        inQueueX.EnQue(xLocal);
        inQueueY.EnQue(yLocal);
        xLocal = inQueueX.DeQue<T>();
        yLocal = inQueueY.DeQue<T>();
        auto zLocal = outQueueZ.AllocTensor<T>();
        AscendC::Add(zLocal, xLocal, yLocal, elementNumPerTile);
        outQueueZ.EnQue(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueY.FreeTensor(yLocal);
        zLocal = outQueueZ.DeQue<T>();
        AscendC::DataCopyPad(zGm[offset], zLocal, copyParams);
        outQueueZ.FreeTensor(zLocal);
    }

    if (tailTileElementNum > 0) {
        int64_t offset = tileNum * elementNumPerTile;
        AscendC::DataCopyExtParams copyParams{1, static_cast<uint32_t>(tailTileElementNum * sizeof(T)), 0, 0, 0};
        AscendC::DataCopyPadExtParams<T> padParams{false, 0, 0, 0};
        auto xLocal = inQueueX.AllocTensor<T>();
        auto yLocal = inQueueY.AllocTensor<T>();
        AscendC::DataCopyPad(xLocal, xGm[offset], copyParams, padParams);
        AscendC::DataCopyPad(yLocal, yGm[offset], copyParams, padParams);
        inQueueX.EnQue(xLocal);
        inQueueY.EnQue(yLocal);
        xLocal = inQueueX.DeQue<T>();
        yLocal = inQueueY.DeQue<T>();
        auto zLocal = outQueueZ.AllocTensor<T>();
        AscendC::Add(zLocal, xLocal, yLocal, tailTileElementNum);
        outQueueZ.EnQue(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueY.FreeTensor(yLocal);
        zLocal = outQueueZ.DeQue<T>();
        AscendC::DataCopyPad(zGm[offset], zLocal, copyParams);
        outQueueZ.FreeTensor(zLocal);
    }
}

// Tiling function
std::tuple<int64_t, int64_t, int64_t> calc_add_tiling_params(int64_t totalLength)
{
    constexpr static int64_t MIN_ELEMS_PER_CORE = 1024;
    constexpr static int64_t BUFFER_NUM = 3;
    auto ascendcPlatform = platform_ascendc::PlatformAscendCManager::GetInstance();
    uint64_t ubSize;
    ascendcPlatform->GetCoreMemSize(platform_ascendc::CoreMemType::UB, ubSize);
    int64_t coreNum = ascendcPlatform->GetCoreNumAiv();
    if (coreNum <= 0) coreNum = 1;
    int64_t numBlocks = std::min(coreNum, (totalLength + MIN_ELEMS_PER_CORE - 1) / MIN_ELEMS_PER_CORE);
    int64_t blockLength = (totalLength + numBlocks - 1) / numBlocks;
    int64_t tileSize = ubSize / PIPELINE_DEPTH / BUFFER_NUM;
    // Floor-align to the 32 B UB block granularity (ops-nn convention). Without
    // this the 6 buffers land 2 bytes under UB on 910B3 (196350 / 196352) -- a
    // margin any allocation padding would blow. 32725 -> 32704 B leaves 128 B.
    tileSize -= tileSize % 32;
    return std::make_tuple(numBlocks, blockLength, tileSize);
}

    auto x_contig = x.contiguous();
    auto idx_contig = dst_indices.contiguous();
    auto out = moe_dispatch_pack_meta(x_contig, idx_contig, total_slots);

    int64_t bs = idx_contig.size(0);
    int64_t k = idx_contig.size(1);
    int64_t hidden_size = x_contig.size(1);
    int64_t total_pairs = bs * k;
    if (total_pairs == 0 || total_slots == 0) return out;

    int64_t numBlocks = std::min(get_moe_pack_aiv_num(), total_pairs);
    auto stream = c10_npu::getCurrentNPUStream().stream(false);
    auto x_ptr = (GM_ADDR)x_contig.data_ptr();
    auto idx_ptr = (GM_ADDR)idx_contig.data_ptr();
    auto out_ptr = (GM_ADDR)out.data_ptr();

    at_npu::native::OpCommand::RunOpApi("MoeDispatchPack", [=]() -> int {
        launch_moe_dispatch_pack_bf16(x_ptr, idx_ptr, out_ptr, bs, k, hidden_size, numBlocks, stream);
        return 0;
    });
    return out;
}

constexpr static int32_t BUFFER_NUM = 2;
constexpr static int64_t MAX_TILE_BYTES = 32768;

__global__ __aicore__ void moe_dispatch_pack_bf16_kernel(
    GM_ADDR x, GM_ADDR dst_indices, GM_ADDR packed_out,
    int64_t bs, int64_t k, int64_t hidden_size)
{
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueue;

    int64_t total_pairs = bs * k;
    int64_t block_idx = AscendC::GetBlockIdx();
    int64_t block_num = AscendC::GetBlockNum();
    int64_t pairs_per_core = (total_pairs + block_num - 1) / block_num;
    int64_t start_pair = block_idx * pairs_per_core;
    int64_t end_pair = ((start_pair + pairs_per_core) < total_pairs ? (start_pair + pairs_per_core) : total_pairs);
    if (start_pair >= end_pair) return;

    int64_t tile_elems = (hidden_size < (MAX_TILE_BYTES / (int64_t)sizeof(bfloat16_t)) ? hidden_size : (MAX_TILE_BYTES / (int64_t)sizeof(bfloat16_t)));
    uint32_t alloc_bytes = static_cast<uint32_t>(tile_elems * sizeof(bfloat16_t));
    pipe.InitBuffer(inQueue, BUFFER_NUM, alloc_bytes);
    pipe.InitBuffer(outQueue, BUFFER_NUM, alloc_bytes);

    AscendC::GlobalTensor<bfloat16_t> xGm, outGm;
    AscendC::GlobalTensor<int32_t> dstIdxGm;
    xGm.SetGlobalBuffer((__gm__ bfloat16_t*)x);
    dstIdxGm.SetGlobalBuffer((__gm__ int32_t*)dst_indices);
    outGm.SetGlobalBuffer((__gm__ bfloat16_t*)packed_out);
    AscendC::DataCopyPadExtParams<bfloat16_t> padParams{false, 0, 0, 0};

    for (int64_t p = start_pair; p < end_pair; ++p) {
        int32_t dst_slot = dstIdxGm.GetValue(p);
        if (dst_slot < 0) continue;
        int64_t src_base = (p / k) * hidden_size;
        int64_t dst_base = static_cast<int64_t>(dst_slot) * hidden_size;

        for (int64_t offset = 0; offset < hidden_size; offset += tile_elems) {
            int64_t cur_elems = ((hidden_size - offset) < tile_elems ? (hidden_size - offset) : tile_elems);
            AscendC::DataCopyExtParams copyParams{1, static_cast<uint32_t>(cur_elems * sizeof(bfloat16_t)), 0, 0, 0};

            auto localIn = inQueue.AllocTensor<bfloat16_t>();
            AscendC::DataCopyPad(localIn, xGm[src_base + offset], copyParams, padParams);
            inQueue.EnQue(localIn);

            localIn = inQueue.DeQue<bfloat16_t>();
            auto localOut = outQueue.AllocTensor<bfloat16_t>();
            AscendC::DataCopy(localOut, localIn, cur_elems);
            outQueue.EnQue(localOut);
            inQueue.FreeTensor(localIn);

            localOut = outQueue.DeQue<bfloat16_t>();
            AscendC::DataCopyPad(outGm[dst_base + offset], localOut, copyParams);
            outQueue.FreeTensor(localOut);
        }
    }
}

__global__ __aicore__ void moe_dispatch_gather_bf16_kernel(
    GM_ADDR recv_buf, GM_ADDR src_indices, GM_ADDR expand_out,
    int64_t total_tokens, int64_t hidden_size)
{
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueue;

    int64_t block_idx = AscendC::GetBlockIdx();
    int64_t block_num = AscendC::GetBlockNum();
    int64_t tokens_per_core = (total_tokens + block_num - 1) / block_num;
    int64_t start_tok = block_idx * tokens_per_core;
    int64_t end_tok = ((start_tok + tokens_per_core) < total_tokens ? (start_tok + tokens_per_core) : total_tokens);
    if (start_tok >= end_tok) return;

    int64_t tile_elems = (hidden_size < (MAX_TILE_BYTES / (int64_t)sizeof(bfloat16_t)) ? hidden_size : (MAX_TILE_BYTES / (int64_t)sizeof(bfloat16_t)));
    uint32_t alloc_bytes = static_cast<uint32_t>(tile_elems * sizeof(bfloat16_t));
    pipe.InitBuffer(inQueue, BUFFER_NUM, alloc_bytes);
    pipe.InitBuffer(outQueue, BUFFER_NUM, alloc_bytes);

    AscendC::GlobalTensor<bfloat16_t> inGm, outGm;
    AscendC::GlobalTensor<int32_t> srcIdxGm;
    inGm.SetGlobalBuffer((__gm__ bfloat16_t*)recv_buf);
    srcIdxGm.SetGlobalBuffer((__gm__ int32_t*)src_indices);
    outGm.SetGlobalBuffer((__gm__ bfloat16_t*)expand_out);
    AscendC::DataCopyPadExtParams<bfloat16_t> padParams{false, 0, 0, 0};

    for (int64_t t = start_tok; t < end_tok; ++t) {
        int32_t src_slot = srcIdxGm.GetValue(t);
        if (src_slot < 0) continue;
        int64_t src_base = static_cast<int64_t>(src_slot) * hidden_size;
        int64_t dst_base = t * hidden_size;

        for (int64_t offset = 0; offset < hidden_size; offset += tile_elems) {
            int64_t cur_elems = ((hidden_size - offset) < tile_elems ? (hidden_size - offset) : tile_elems);
            AscendC::DataCopyExtParams copyParams{1, static_cast<uint32_t>(cur_elems * sizeof(bfloat16_t)), 0, 0, 0};

            auto localIn = inQueue.AllocTensor<bfloat16_t>();
            AscendC::DataCopyPad(localIn, inGm[src_base + offset], copyParams, padParams);
            inQueue.EnQue(localIn);

            localIn = inQueue.DeQue<bfloat16_t>();
            auto localOut = outQueue.AllocTensor<bfloat16_t>();
            AscendC::DataCopy(localOut, localIn, cur_elems);
            outQueue.EnQue(localOut);
            inQueue.FreeTensor(localIn);

            localOut = outQueue.DeQue<bfloat16_t>();
            AscendC::DataCopyPad(outGm[dst_base + offset], localOut, copyParams);
            outQueue.FreeTensor(localOut);
        }
    }
}

static torch::Tensor bg_pack_pool;
static torch::Tensor bg_max_pool;
static torch::Tensor bg_expand_pool;
static torch::Tensor bg_cnt_pool;

std::tuple<torch::Tensor, torch::Tensor> moe_bg_pack_npu(const torch::Tensor &x, const torch::Tensor &expert_ids, int64_t ep_world_size, int64_t moe_expert_num, int64_t stride_rows)
{
    const c10::OptionalDeviceGuard guard(x.device());
    auto x_c = x.contiguous(); auto idx_c = expert_ids.contiguous();
    int64_t req_rows = ep_world_size * stride_rows, h = x_c.size(1);
    if (!bg_pack_pool.defined() || bg_pack_pool.device() != x.device() || bg_pack_pool.size(0) < req_rows || bg_pack_pool.size(1) != h) {
        bg_pack_pool = torch::empty({req_rows, h}, x.options());
    }
    if (!bg_max_pool.defined() || bg_max_pool.device() != x.device()) {
        bg_max_pool = torch::empty({1}, idx_c.options().dtype(torch::kInt32));
    }
    auto out = bg_pack_pool.narrow(0, 0, req_rows);
    int64_t bs = idx_c.size(0), k = idx_c.size(1);
    int64_t numBlocks = std::min(get_bg_aiv_num(), bs * k);
    auto stream = c10_npu::getCurrentNPUStream().stream(false);
    auto xp = (GM_ADDR)x_c.data_ptr(), ip = (GM_ADDR)idx_c.data_ptr(), op = (GM_ADDR)out.data_ptr(), mp = (GM_ADDR)bg_max_pool.data_ptr();
    at_npu::native::OpCommand::RunOpApi("MoeBgPack", [=]() -> int { launch_moe_bg_pack_bf16(xp, ip, op, mp, bs, k, h, ep_world_size, moe_expert_num, stride_rows, numBlocks, stream); return 0; });
    return std::make_tuple(out, bg_max_pool);
}

#include "kernel_operator.h"
#include "platform/platform_ascendc.h"
#include "moe_beat_golden_launch.h"

constexpr static int32_t BUFFER_NUM = 2;
constexpr static int64_t MAX_TILE_BYTES = 65536; // 单队列可安全独享 64KB 双缓冲 (128KB/192KB)
constexpr static int64_t MAX_PAIRS_UB = 8192;
constexpr static int32_t MAX_EXPERTS = 128;
constexpr static int32_t MAX_RANKS = 16;

__global__ __aicore__ void moe_bg_pack_bf16_kernel(
    GM_ADDR x, GM_ADDR expert_ids, GM_ADDR packed_out, GM_ADDR max_rows_out,
    int64_t bs, int64_t k, int64_t hidden_size,
    int64_t ep_world_size, int64_t moe_expert_num, int64_t stride_rows)
{
    AscendC::TPipe pipe;
    AscendC::TQueBind<AscendC::QuePosition::VECIN, AscendC::QuePosition::VECOUT, BUFFER_NUM> dmaQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> idxBuf;

    int64_t total_pairs = bs * k;
    int64_t block_idx = AscendC::GetBlockIdx();
    int64_t block_num = AscendC::GetBlockNum();
    int64_t pairs_per_core = (total_pairs + block_num - 1) / block_num;
    int64_t start_pair = block_idx * pairs_per_core;
    int64_t end_pair = (start_pair + pairs_per_core) < total_pairs ? (start_pair + pairs_per_core) : total_pairs;
    if (start_pair >= end_pair) return;

    int64_t max_elems = MAX_TILE_BYTES / (int64_t)sizeof(bfloat16_t);
    int64_t tile_elems = hidden_size < max_elems ? hidden_size : max_elems;
    uint32_t alloc_bytes = static_cast<uint32_t>(tile_elems * sizeof(bfloat16_t));
    pipe.InitBuffer(dmaQueue, BUFFER_NUM, alloc_bytes);
    pipe.InitBuffer(idxBuf, MAX_PAIRS_UB * sizeof(int32_t));

    AscendC::GlobalTensor<bfloat16_t> xGm, outGm;
    AscendC::GlobalTensor<int32_t> expIdGm;
    xGm.SetGlobalBuffer((__gm__ bfloat16_t*)x);
    expIdGm.SetGlobalBuffer((__gm__ int32_t*)expert_ids);
    outGm.SetGlobalBuffer((__gm__ bfloat16_t*)packed_out);

    auto localExp = idxBuf.Get<int32_t>();
    int64_t aligned_pairs = ((total_pairs + 7) / 8) * 8;
    if (aligned_pairs > MAX_PAIRS_UB) aligned_pairs = MAX_PAIRS_UB;
    AscendC::DataCopyExtParams idxCopy{1, static_cast<uint32_t>(total_pairs * sizeof(int32_t)), 0, 0, 0};
    AscendC::DataCopyPadExtParams<int32_t> idxPad{true, 0, static_cast<uint8_t>(aligned_pairs - total_pairs), 0};
    AscendC::DataCopyPad(localExp, expIdGm, idxCopy, idxPad);
    AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
    AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);

    int32_t experts_per_rank = static_cast<int32_t>(moe_expert_num / ep_world_size);
    int32_t exp_counts[MAX_EXPERTS];
    int32_t prefix_before[MAX_EXPERTS];
    for (int32_t e = 0; e < moe_expert_num; ++e) { exp_counts[e] = 0; prefix_before[e] = 0; }

    for (int64_t p = 0; p < total_pairs; ++p) {
        int32_t eid = localExp.GetValue(p);
        if (p < start_pair) prefix_before[eid]++;
        exp_counts[eid]++;
    }

    int32_t exp_dst_base[MAX_EXPERTS];
    int32_t max_rank_rows = 0;
    for (int32_t r = 0; r < ep_world_size; ++r) {
        int32_t acc = 0;
        for (int32_t le = 0; le < experts_per_rank; ++le) {
            int32_t gid = r * experts_per_rank + le;
            exp_dst_base[gid] = static_cast<int32_t>(r * stride_rows) + acc;
            acc += exp_counts[gid];
        }
        if (acc > max_rank_rows) max_rank_rows = acc;
        if (block_idx == 0) {
            __gm__ int32_t* meta_ptr = reinterpret_cast<__gm__ int32_t*>(
                reinterpret_cast<__gm__ bfloat16_t*>(packed_out) + (r * stride_rows + (stride_rows - 1)) * hidden_size);
            for (int32_t le = 0; le < experts_per_rank; ++le) meta_ptr[le] = exp_counts[r * experts_per_rank + le];
        }
    }
    if (block_idx == 0) {
        reinterpret_cast<__gm__ int32_t*>(max_rows_out)[0] = max_rank_rows;
    }

    // 使用 TQueBind 直通队列 + 原生 32B 对齐 DataCopy 突发传输（绕过 Pad 状态机，零 UB 冗余拷贝！）
    for (int64_t p = start_pair; p < end_pair; ++p) {
        int32_t eid = localExp.GetValue(p);
        int32_t dst_slot = exp_dst_base[eid] + (prefix_before[eid]++);
        int64_t src_base = (p / k) * hidden_size;
        int64_t dst_base = static_cast<int64_t>(dst_slot) * hidden_size;

        for (int64_t offset = 0; offset < hidden_size; offset += tile_elems) {
            int64_t cur_elems = (hidden_size - offset) < tile_elems ? (hidden_size - offset) : tile_elems;
            auto buf = dmaQueue.AllocTensor<bfloat16_t>();
            AscendC::DataCopy(buf, xGm[src_base + offset], cur_elems);
            dmaQueue.EnQue(buf);
            buf = dmaQueue.DeQue<bfloat16_t>();
            AscendC::DataCopy(outGm[dst_base + offset], buf, cur_elems);
            dmaQueue.FreeTensor(buf);
        }
    }
}

__global__ __aicore__ void moe_bg_unpack_bf16_kernel(
    GM_ADDR recv_buf, GM_ADDR expand_out, GM_ADDR ep_recv_counts,
    int64_t hidden_size, int64_t ep_world_size, int64_t moe_expert_num, int64_t stride_rows)
{
    AscendC::TPipe pipe;
    AscendC::TQueBind<AscendC::QuePosition::VECIN, AscendC::QuePosition::VECOUT, BUFFER_NUM> dmaQueue;

    int64_t block_idx = AscendC::GetBlockIdx();
    int64_t block_num = AscendC::GetBlockNum();
    int64_t max_elems = MAX_TILE_BYTES / (int64_t)sizeof(bfloat16_t);
    int64_t tile_elems = hidden_size < max_elems ? hidden_size : max_elems;
    uint32_t alloc_bytes = static_cast<uint32_t>(tile_elems * sizeof(bfloat16_t));
    pipe.InitBuffer(dmaQueue, BUFFER_NUM, alloc_bytes);

    AscendC::GlobalTensor<bfloat16_t> inGm, outGm;
    inGm.SetGlobalBuffer((__gm__ bfloat16_t*)recv_buf);
    outGm.SetGlobalBuffer((__gm__ bfloat16_t*)expand_out);

    int32_t experts_per_rank = static_cast<int32_t>(moe_expert_num / ep_world_size);
    int32_t recv_cnt[MAX_RANKS][MAX_EXPERTS];
    int32_t src_off[MAX_RANKS][MAX_EXPERTS];

    for (int32_t r = 0; r < ep_world_size; ++r) {
        __gm__ int32_t* meta_ptr = reinterpret_cast<__gm__ int32_t*>(
            reinterpret_cast<__gm__ bfloat16_t*>(recv_buf) + (r * stride_rows + (stride_rows - 1)) * hidden_size);
        int32_t acc = static_cast<int32_t>(r * stride_rows);
        for (int32_t le = 0; le < experts_per_rank; ++le) {
            int32_t c = meta_ptr[le];
            recv_cnt[r][le] = c;
            src_off[r][le] = acc;
            acc += c;
        }
    }

    int32_t dst_off[MAX_EXPERTS][MAX_RANKS];
    int32_t total_recv_tokens = 0;
    for (int32_t le = 0; le < experts_per_rank; ++le) {
        int32_t exp_sum = 0;
        for (int32_t r = 0; r < ep_world_size; ++r) {
            dst_off[le][r] = total_recv_tokens;
            total_recv_tokens += recv_cnt[r][le];
            exp_sum += recv_cnt[r][le];
        }
        if (block_idx == 0) reinterpret_cast<__gm__ int32_t*>(ep_recv_counts)[le] = exp_sum;
    }
    if (total_recv_tokens == 0) return;

    int64_t tokens_per_core = (total_recv_tokens + block_num - 1) / block_num;
    int64_t start_tok = block_idx * tokens_per_core;
    int64_t end_tok = (start_tok + tokens_per_core) < total_recv_tokens ? (start_tok + tokens_per_core) : total_recv_tokens;
    if (start_tok >= end_tok) return;

    for (int32_t le = 0; le < experts_per_rank; ++le) {
        for (int32_t r = 0; r < ep_world_size; ++r) {
            int32_t cnt = recv_cnt[r][le];
            if (cnt <= 0) continue;
            int64_t blk_dst_start = dst_off[le][r];
            int64_t ov_start = blk_dst_start > start_tok ? blk_dst_start : start_tok;
            int64_t ov_end = (blk_dst_start + cnt) < end_tok ? (blk_dst_start + cnt) : end_tok;
            for (int64_t t = ov_start; t < ov_end; ++t) {
                int64_t src_base = (static_cast<int64_t>(src_off[r][le]) + (t - blk_dst_start)) * hidden_size;
                int64_t dst_base = t * hidden_size;
                for (int64_t offset = 0; offset < hidden_size; offset += tile_elems) {
                    int64_t cur_elems = (hidden_size - offset) < tile_elems ? (hidden_size - offset) : tile_elems;
                    auto buf = dmaQueue.AllocTensor<bfloat16_t>();
                    AscendC::DataCopy(buf, inGm[src_base + offset], cur_elems);
                    dmaQueue.EnQue(buf);
                    buf = dmaQueue.DeQue<bfloat16_t>();
                    AscendC::DataCopy(outGm[dst_base + offset], buf, cur_elems);
                    dmaQueue.FreeTensor(buf);
                }
            }
        }
    }
}

    int64_t coreNum = platform->GetCoreNumAiv();
    return coreNum > 0 ? coreNum : 20;
}

torch::Tensor sqrt_npu(const torch::Tensor &x)
{
    const c10::OptionalDeviceGuard guard(x.device());
    auto z = sqrt_meta(x);
    auto stream = c10_npu::getCurrentNPUStream().stream(false);
    int64_t totalLength = x.numel();
    int64_t numBlocks, blockLength, tileSize;
    std::tie(numBlocks, blockLength, tileSize) = calc_sqrt_tiling_params(totalLength);
    auto x_ptr = (GM_ADDR)x.data_ptr();
    auto z_ptr = (GM_ADDR)z.data_ptr();

    auto acl_call = [=]() -> int {
        auto dtype = x.scalar_type();
        if      (dtype == torch::kFloat32) launch_sqrt_kernel_float   (x_ptr, z_ptr, totalLength, numBlocks, blockLength, tileSize, stream);
        else if (dtype == torch::kFloat16) launch_sqrt_kernel_half    (x_ptr, z_ptr, totalLength, numBlocks, blockLength, tileSize, stream);
        else if (dtype == torch::kBFloat16) launch_sqrt_kernel_bfloat16(x_ptr, z_ptr, totalLength, numBlocks, blockLength, tileSize, stream);
        return 0;
    };
    at_npu::native::OpCommand::RunOpApi("Sqrt", acl_call);
    return z;
}

/*!
 * \file sqrt_host.cpp
 * \brief Sqrt host code - kernel launch and tiling (compiled with bisheng + -xasc)
 *
 * AscendC::Sqrt has no native fp16/bf16 overload, so for those dtypes we
 * cast the input to fp32, run Sqrt, and cast the result back.
 */

#include <tuple>
#include <algorithm>
#include <type_traits>
#include "kernel_operator.h"
#include "platform/platform_ascendc.h"

constexpr static int64_t PIPELINE_DEPTH = 2;

template <typename T>
class KernelSqrt {
public:
    __aicore__ inline KernelSqrt() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, int64_t totalLength, int64_t blockLength, uint32_t tileSize)
    {
        xGm_.SetGlobalBuffer((__gm__ T *)x + blockLength * AscendC::GetBlockIdx());
        zGm_.SetGlobalBuffer((__gm__ T *)z + blockLength * AscendC::GetBlockIdx());
        // tileSize is element count (not bytes)
        pipe_.InitBuffer(inQueueX_,  PIPELINE_DEPTH, tileSize * sizeof(T));
        pipe_.InitBuffer(outQueueZ_, PIPELINE_DEPTH, tileSize * sizeof(T));
        if constexpr (!std::is_same<T, float>::value) {
            pipe_.InitBuffer(xFloatBuf_, tileSize * sizeof(float));
            pipe_.InitBuffer(zFloatBuf_, tileSize * sizeof(float));
        }
        int64_t currentBlockLength = totalLength - AscendC::GetBlockIdx() * blockLength;
        if (currentBlockLength > blockLength) currentBlockLength = blockLength;
        if (currentBlockLength < 0) currentBlockLength = 0;
        elementNumPerTile_ = tileSize;
        tileNum_ = currentBlockLength / elementNumPerTile_;
        tailTileElementNum_ = currentBlockLength - tileNum_ * elementNumPerTile_;
    }

    __aicore__ inline void Process()
    {
        for (int64_t i = 0; i < tileNum_; ++i) {
            CopyIn(i * elementNumPerTile_, elementNumPerTile_);
            Compute(elementNumPerTile_);
            CopyOut(i * elementNumPerTile_, elementNumPerTile_);
        }
        if (tailTileElementNum_ > 0) {
            CopyIn(tileNum_ * elementNumPerTile_, tailTileElementNum_);
            Compute(tailTileElementNum_);
            CopyOut(tileNum_ * elementNumPerTile_, tailTileElementNum_);
        }
    }

private:
    __aicore__ inline void CopyIn(int64_t offset, int64_t count)
    {
        AscendC::DataCopyExtParams copyParams{1, static_cast<uint32_t>(count * sizeof(T)), 0, 0, 0};
        AscendC::DataCopyPadExtParams<T> padParams{false, 0, 0, 0};
        auto xLocal = inQueueX_.AllocTensor<T>();
        AscendC::DataCopyPad(xLocal, xGm_[offset], copyParams, padParams);
        inQueueX_.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int64_t count)
    {
        auto xLocal = inQueueX_.DeQue<T>();
        auto zLocal = outQueueZ_.AllocTensor<T>();
        if constexpr (std::is_same<T, float>::value) {
            AscendC::Sqrt(zLocal, xLocal, count);
        } else {
            auto xF = xFloatBuf_.Get<float>();
            auto zF = zFloatBuf_.Get<float>();
            AscendC::Cast(xF, xLocal, AscendC::RoundMode::CAST_NONE, count);
            AscendC::Sqrt(zF, xF, count);
            constexpr auto roundMode = std::is_same<T, bfloat16_t>::value
                ? AscendC::RoundMode::CAST_RINT : AscendC::RoundMode::CAST_NONE;
            AscendC::Cast(zLocal, zF, roundMode, count);
        }
        outQueueZ_.EnQue(zLocal);
        inQueueX_.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int64_t offset, int64_t count)
    {
        auto zLocal = outQueueZ_.DeQue<T>();
        AscendC::DataCopyExtParams copyParams{1, static_cast<uint32_t>(count * sizeof(T)), 0, 0, 0};
        AscendC::DataCopyPad(zGm_[offset], zLocal, copyParams);
        outQueueZ_.FreeTensor(zLocal);
    }

    AscendC::TPipe pipe_;
    AscendC::GlobalTensor<T> xGm_, zGm_;
    AscendC::TQue<AscendC::TPosition::VECIN, PIPELINE_DEPTH> inQueueX_;
    AscendC::TQue<AscendC::TPosition::VECOUT, PIPELINE_DEPTH> outQueueZ_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xFloatBuf_, zFloatBuf_;
    int64_t elementNumPerTile_ = 0, tileNum_ = 0, tailTileElementNum_ = 0;
};

template <typename T>
__global__ __aicore__ __vector__ void sqrt_kernel(GM_ADDR x, GM_ADDR z, int64_t totalLength, int64_t blockLength, uint32_t tileSize)
{
    KernelSqrt<T> op;
    op.Init(x, z, totalLength, blockLength, tileSize);
    op.Process();
}

// Returns (numBlocks, blockLength, tileSize) where tileSize is element count
// per UB tile, derived at runtime from the UB capacity.
//
// Unlike add, the per-element UB cost here depends on T (queue depth
// included): fp32 needs only the in/out queues, while fp16/bf16 also need the
// two fp32 cast buffers. The tiling function cannot see T, so budget for the
// worst case -- fp16/bf16 -- which is also safe (just a smaller tile) for fp32.
std::tuple<int64_t, int64_t, int64_t> calc_sqrt_tiling_params(int64_t totalLength)
{
    constexpr static int64_t MIN_ELEMS_PER_CORE = 1024;
    // Full per-element UB cost, queue depth already folded in (see Init()):
    //   fp32      -- in/out queues: 2 * PIPELINE_DEPTH * 4 B            = 16 B
    //   fp16/bf16 -- in/out queues: 2 * PIPELINE_DEPTH * 2 B    =  8 B,
    //                plus the two fp32 cast buffers 2 * 4 B           =  8 B
    // Both paths land on 16 B per element, so a single depth-inclusive tile
    // budget covers every dtype. Do NOT divide by PIPELINE_DEPTH again below;
    // it is already part of these terms.
    constexpr static int64_t BYTES_PER_ELEM_FP32 = 2 * PIPELINE_DEPTH * sizeof(float);
    constexpr static int64_t BYTES_PER_ELEM_FP16 =
        2 * PIPELINE_DEPTH * sizeof(uint16_t) + 2 * sizeof(float);
    constexpr static int64_t BYTES_PER_ELEM_MAX =
        std::max(BYTES_PER_ELEM_FP32, BYTES_PER_ELEM_FP16);

    auto ascendcPlatform = platform_ascendc::PlatformAscendCManager::GetInstance();
    uint64_t ubSize = 0;
    ascendcPlatform->GetCoreMemSize(platform_ascendc::CoreMemType::UB, ubSize);
    int64_t coreNum = ascendcPlatform->GetCoreNumAiv();
    if (coreNum <= 0) coreNum = 1;
    int64_t numBlocks = std::min(coreNum, (totalLength + MIN_ELEMS_PER_CORE - 1) / MIN_ELEMS_PER_CORE);
    numBlocks = std::max(numBlocks, static_cast<int64_t>(1));
    int64_t blockLength = (totalLength + numBlocks - 1) / numBlocks;
    int64_t tileSize = static_cast<int64_t>(ubSize) / BYTES_PER_ELEM_MAX;
    // Enforce strict 32 B alignment on every InitBuffer size (the GetUbBlockSize()
    // granularity ops-nn aligns to): the narrowest supported T is fp16 at 2 B, so
    // flooring the element count to 32 B / 2 B = 16 makes tileSize * sizeof(T) an
    // exact multiple of 32 B for every dtype (fp16 buffers land exactly on 32 B,
    // the 4 B fp32/cast buffers on 64 B) and TPipe applies no padding.
    // A no-op on 910B3: 196352 / 16 = 12272 = 16 * 767, so the aligned tile
    // fills the UB exactly (196352 B allocated, ~100% utilization).
    constexpr static int64_t TILE_ALIGN_ELEMS = 32 / sizeof(uint16_t);
    tileSize -= tileSize % TILE_ALIGN_ELEMS;
    // A zero tileSize would divide by zero in Init().
    if (tileSize < 1) tileSize = 1;
    return std::make_tuple(numBlocks, blockLength, tileSize);
}
